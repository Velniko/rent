function rosterAllowedWindows(day,ranges,business){
  if(!rosterDateValid(day))return [];
  let windows=ranges.map(r=>({start:rosterMinutes(r.start),end:rosterMinutes(r.end)})).filter(w=>w.end>w.start);
  const weekday=new Date(day+'T12:00:00Z').getUTCDay();
  const closed=(business.unavailablePickupHours||[]).filter(r=>Number(r.weekday)===weekday).map(r=>({start:rosterMinutes(r.start),end:rosterMinutes(r.end)}));
  for(const range of business.unavailablePickupRanges||[]){
    if(range.start.slice(0,10)>day||range.end.slice(0,10)<day)continue;
    closed.push({start:range.start.slice(0,10)===day?rosterMinutes(range.start.slice(11,16)):0,end:range.end.slice(0,10)===day?rosterMinutes(range.end.slice(11,16)):1440});
  }
  for(const block of closed)windows=windows.flatMap(w=>!rosterOverlap(w,block)?[w]:[{start:w.start,end:Math.min(w.end,block.start)},{start:Math.max(w.start,block.end),end:w.end}].filter(w=>w.end>w.start));
  return windows.sort((a,b)=>a.start-b.start);
}
const rosterOpenWindows=(day,business)=>rosterAllowedWindows(day,[business.businessHours],business);

function rosterTaskMode(task,business){
  const start=rosterMinutes(task.start),end=rosterMinutes(task.end);if(end<=start)return null;
  if(rosterOpenWindows(task.day,business).some(w=>start>=w.start&&end<=w.end))return 'business';
  if(rosterServiceWindows(task.day,business).some(w=>start>=w.start&&end<=w.end))return 'service';
  return null;
}
function rosterCoverageWindows(day,tasks,business){
  const normal=rosterOpenWindows(day,business).map(w=>({...w,service:false}));
  const booked=tasks.filter(t=>t.day===day&&rosterTaskMode(t,business)==='service').map(t=>({start:rosterMinutes(t.start),end:rosterMinutes(t.end),service:true})).sort((a,b)=>a.start-b.start);
  const merged=[];for(const window of booked){const last=merged.at(-1);if(last&&window.start<=last.end)last.end=Math.max(last.end,window.end);else merged.push({...window})}
  return normal.concat(merged).sort((a,b)=>a.start-b.start);
}

function rosterCanOperate({employeeId,task,shifts,grants,business,now=ROSTER_NOW,role='STAFF'}){
  if(role==='ADMIN')return {allowed:true,via:'管理员'};
  if(!task)return {allowed:false,reason:'任务不存在'};
  const day=now.slice(0,10),time=rosterMinutes(now.slice(11,16));if(day!==task.day)return {allowed:false,reason:'只可办理当天任务'};
  const inBusiness=rosterOpenWindows(day,business).some(w=>time>=w.start&&time<w.end);
  const inBookedService=rosterTaskMode(task,business)==='service'&&time>=rosterMinutes(task.start)&&time<rosterMinutes(task.end);
  if(!inBusiness&&!inBookedService)return {allowed:false,reason:'当前不在营业或预约服务时段'};
  const grant=(grants||[]).find(g=>g.employeeId===employeeId&&g.status==='APPROVED'&&g.day===day&&time>=rosterMinutes(g.start)&&time<rosterMinutes(g.end)&&(g.scope==='all'||g.taskId===task.id));
  if(grant)return {allowed:true,via:'临时授权',grantId:grant.id};
  if(!shifts.some(s=>s.employeeId===employeeId&&s.day===day&&time>=rosterMinutes(s.start)&&time<rosterMinutes(s.end)))return {allowed:false,reason:'当前未排班'};
  if(task.employeeId!==employeeId)return {allowed:false,reason:'任务已分配给其他员工'};
  return {allowed:true,via:'排班'};
}

}

function assignRosterTasks(tasks,shifts,business,overrides={}){
  const assigned=[];
  for(const task of [...tasks].sort((a,b)=>a.day.localeCompare(b.day)||Number(Boolean(overrides[b.id]))-Number(Boolean(overrides[a.id]))||a.start.localeCompare(b.start)||a.id.localeCompare(b.id))){
    const start=rosterMinutes(task.start),end=rosterMinutes(task.end),mode=rosterTaskMode(task,business);
    if(!mode){assigned.push({...task,employeeId:null,assignmentReason:'不在可预约取还时段'});continue}
    const eligible=[...new Set(shifts.filter(s=>s.day===task.day&&rosterMinutes(s.start)<=start&&rosterMinutes(s.end)>=end).map(s=>s.employeeId))].filter(id=>!assigned.some(t=>t.employeeId===id&&t.day===task.day&&rosterOverlap({start:rosterMinutes(t.start),end:rosterMinutes(t.end)},{start,end})));
    eligible.sort((a,b)=>Number(b===overrides[task.id])-Number(a===overrides[task.id])||assigned.filter(t=>t.employeeId===a&&t.day===task.day).length-assigned.filter(t=>t.employeeId===b&&t.day===task.day).length||a.localeCompare(b));
    assigned.push({...task,service:mode==='service',employeeId:eligible[0]||null,assignmentReason:eligible.length?'':'无可用当班员工'});
  }
  return assigned.sort((a,b)=>a.day.localeCompare(b.day)||a.start.localeCompare(b.start)||a.id.localeCompare(b.id));
}

  rosterRoot.innerHTML=`<section class="panel"><div class="roster-log-toolbar"><h3>排班与临时操作记录</h3><label class="roster-log-search"><input class="form-control" type="search" data-roster-log-search value="${rosterEscape(rosterLogSearch)}" placeholder="搜索员工、操作或订单" aria-label="搜索排班操作记录"></label></div><div class="roster-log-list">${logs.map(l=>`<article><span class="roster-log-dot"></span><div class="roster-log-when mono">${l.at.replace('T',' ')}</div><div class="roster-log-content"><strong>${rosterEscape(l.action)}</strong><p>${rosterEscape(l.target)}</p><span>${rosterEscape(l.detail)}</span></div><span class="roster-log-actor">${rosterEscape(l.actor)}</span></article>`).join('')||'<div class="empty-state">没有找到记录</div>'}</div></section>`;
}

function rosterOpenPreference(employeeId){
  const employee=rosterEmployee(employeeId);if(!employee)return;
  const permission=rosterPreferencePermission(employee,{role:rosterIsAdmin?'ADMIN':'STAFF',actorId:rosterState.currentEmployeeId,now:rosterState.instant,policy:rosterState.preferencePolicy});
  const form=document.getElementById('rosterPreferenceForm');form.reset();form.elements.employeeId.value=employeeId;form.elements.start.value=employee.preferredStart||'';form.elements.end.value=employee.preferredEnd||'';
  form.querySelector('[data-roster-preference-person]').innerHTML=`<div class="roster-decision-person"><i class="roster-avatar">${rosterEscape(employee.name[0])}</i><div><strong>${rosterEscape(employee.name)}</strong><span>当前偏好：${employee.preferredStart?employee.preferredStart+'–'+employee.preferredEnd:'不限时段'}</span></div></div>`;
  form.querySelector('[data-roster-preference-rule]').textContent=rosterIsAdmin?'管理员可随时修改':permission.allowed?'每30天可修改一次。':`下次可修改：${rosterLocalDate(permission.nextAllowedAt).replaceAll('-','.')}`;
  form.querySelector('button[type="submit"]').disabled=!permission.allowed;form.elements.start.disabled=!permission.allowed;form.elements.end.disabled=!permission.allowed;rosterOpenDialog('.roster-dialog--preference');
}

function rosterNotify(message){window.workspaceNotify(message)}
async function rosterCommand(operation,fields={},reason='update'){
 const response=await fetch((rosterIsAdmin?'/admin':'/staff')+'/roster/action',{method:'POST',credentials:'same-origin',headers:{'Content-Type':'application/json','Accept':'application/json'},body:JSON.stringify({operation,revision:rosterState.revision,startDate:rosterState.selectedStart,...fields})});
 const result=await response.json();if(!response.ok)throw new Error(result.error||'操作失败，请重试');
 rosterState=result.data;ROSTER_TODAY=rosterState.today;ROSTER_NOW=rosterState.now;rosterRender(reason);return result;
}
async function rosterPeriod(start){const response=await fetch((rosterIsAdmin?'/admin':'/staff')+'/roster/data?start='+encodeURIComponent(start),{headers:{Accept:'application/json'}});const result=await response.json();if(!response.ok)throw new Error(result.error||'加载失败');rosterState=result.data;ROSTER_TODAY=rosterState.today;ROSTER_NOW=rosterState.now;rosterRender('week')}
function rosterBindForm(id,operation,fields,reason='update'){
 const form=document.getElementById(id);if(!form)return;
 form.addEventListener('submit',async event=>{event.preventDefault();if(form.dataset.motionBusy||form.closest('dialog').classList.contains('motion-closing'))return;rosterError(form,'');if(!form.reportValidity())return;const data=new FormData(form);form.dataset.motionBusy='true';try{await window.WorkspaceMotion.task(event.submitter,async()=>{await rosterCommand(operation,fields(data),reason);await window.WorkspaceMotion.close(form.closest('dialog'));rosterNotify('已保存')})}catch(error){rosterError(form,error.message)}finally{delete form.dataset.motionBusy}});
}
