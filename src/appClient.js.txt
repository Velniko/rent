/* 站点前端脚本。原先内联在 layout.html 里，每次整页加载都要重新下载 + 解析；
 * 抽成带内容指纹的外部文件后可长期缓存、只解析一次。SPA 局部导航只替换 .content，
 * 不受影响。以 defer 加载，执行时 DOM 已就绪；不启用 strict 模式以保持与原内联脚本一致。 */
(function () {
  // 成功反馈轻量展示；错误和业务警告保留在原处，直到用户处理或关闭。
  function getNoticeLayer() {
    var layer = document.getElementById('site-notice-layer');
    if (!layer) {
      layer = document.createElement('div');
      layer.id = 'site-notice-layer';
      layer.className = 'notification-toast-layer';
      layer.setAttribute('aria-label', '操作反馈与新通知');
      document.body.appendChild(layer);
    }
    return layer;
  }
  function addNoticeClose(notice) {
    var close = document.createElement('button');
    close.type = 'button';
    close.className = 'notice-close';
    close.setAttribute('aria-label', '关闭提示');
    close.textContent = '×';
    close.addEventListener('click', function () { notice.dismissNotice ? notice.dismissNotice() : notice.remove(); });
    notice.classList.add('notice-dismissible');
    notice.appendChild(close);
  }
  window.siteToast = function (message, type) {
    var notice = document.createElement('div');
    var failed = type === 'error';
    notice.className = 'alert alert-' + (failed ? 'error' : 'success') + ' site-toast';
    notice.setAttribute('role', failed ? 'alert' : 'status');
    notice.setAttribute('aria-atomic', 'true');
    notice.textContent = String(message || '');
    notice.dataset.toastType = failed ? 'error' : 'success';
    var timer;
    notice.dismissNotice = function () { window.clearTimeout(timer); notice.remove(); };
    addNoticeClose(notice);
    var layer = getNoticeLayer();
    var previous = Array.from(layer.querySelectorAll('.site-toast')).filter(function (item) { return item.dataset.toastType === 'success'; });
    if (!failed && previous.length >= 3) previous[0].dismissNotice();
    layer.appendChild(notice);
    if (!failed) timer = window.setTimeout(notice.dismissNotice, 5000);
    return notice;
  };
  function prepareFlashNotices(isReload) {
    var notices = document.querySelectorAll('[data-flash-notice]');
    if (!notices.length) return;
    var url = new URL(window.location.href);
    ['success', 'error', 'created', 'reset', 'deletion_requested', 'passwordSetup'].forEach(function (key) {
      url.searchParams.delete(key);
    });
    if (url.href !== window.location.href) window.history.replaceState(window.history.state, '', url.href);
    notices.forEach(function (notice) {
      if (notice.dataset.flashReady === 'true') return;
      notice.dataset.flashReady = 'true';
      // 高危提示由页面明确标记，不能被当作短暂操作反馈隐藏。
      if (notice.dataset.noticePriority === 'critical') return;
      var success = notice.matches('.page-notification--success, .alert-success') || notice.querySelector('.alert-success');
      if (success && !notice.querySelector('a, button, input, form')) {
        if (!isReload) window.siteToast(notice.textContent, 'success');
        notice.remove();
        return;
      }
      notice.setAttribute('role', notice.matches('.page-notification--error, .alert-error, .alert-danger') ? 'alert' : 'status');
      notice.setAttribute('aria-atomic', 'true');
      addNoticeClose(notice);
    });
  }
  var navigation = window.performance && window.performance.getEntriesByType('navigation')[0];
  prepareFlashNotices(navigation && navigation.type === 'reload');

  function prepareImageUploadForms() {
    document.querySelectorAll('form').forEach(function (form) {
      var proofInput = form.querySelector('input[name="imageUrl"]');
      var contractProofInput = form.querySelector('input[name="transferProofUrl"]');
      var damageInput = form.querySelector('textarea[name="damagePhotos"]');
      var damageFileInput = form.querySelector('input[name="damagePhotoFiles"]');
      if (!proofInput && !contractProofInput && !damageInput && !damageFileInput) return;
      form.enctype = 'multipart/form-data';
      var input = proofInput || contractProofInput;
      if (input) {
        input.type = 'file';
        input.name = proofInput ? 'imageFile' : 'transferProofFile';
        input.accept = 'image/jpeg,image/png,image/webp';
        input.removeAttribute('placeholder');
        var label = input.previousElementSibling;
        if (label && label.tagName === 'LABEL') label.textContent = '付款凭证图片';
      }
      if (damageInput) {
        var fileInput = document.createElement('input');
        fileInput.className = damageInput.className || 'form-control';
        fileInput.type = 'file';
        fileInput.name = 'damagePhotoFiles';
        fileInput.accept = 'image/jpeg,image/png,image/webp';

        var damageLabel = fileInput.previousElementSibling;
        if (damageLabel && damageLabel.tagName === 'LABEL') damageLabel.textContent = '损坏照片';
      }
    });
  }

    window.createRichTextEditor = function (selector, options) {
      if (window.Quill) {
        try {
          return new window.Quill(selector, options);
        } catch (error) {
          console.error('Quill initialization failed, falling back to contentEditable:', error);
        }
      }
      const root = document.querySelector(selector);
      if (!root) throw new Error('Rich text editor element not found: ' + selector);
      root.contentEditable = 'true';
      root.setAttribute('role', 'textbox');
      root.setAttribute('aria-multiline', 'true');
      root.classList.add('rich-text-fallback');
      const toolbar = document.createElement('div');
      toolbar.className = 'rich-text-fallback-toolbar';
      const commands = [
        ['正文', 'formatBlock', 'p'], ['H1', 'formatBlock', 'h1'], ['H2', 'formatBlock', 'h2'],
        ['B', 'bold'], ['I', 'italic'], ['U', 'underline'], ['S', 'strike'],
        ['1.', 'insertOrderedList'], ['•', 'insertUnorderedList'], ['“”', 'formatBlock', 'blockquote'],
        ['代码', 'formatBlock', 'pre'], ['颜色', 'foreColor'], ['背景', 'hiliteColor'],

          event.preventDefault();
          root.focus();

          if (item[1] === 'createLink') {

              root.focus();

            return;
          }
          document.execCommand(item[1], false, item[2] || null);
        });
        toolbar.appendChild(button);
      });

      return { root: root };
    };

    window.markdownToHtml = function (markdown) {
      if (window.marked && typeof window.marked.parse === 'function') {
        return window.marked.parse(String(markdown || ''), { breaks: true, gfm: true });
      }
      if (typeof window.marked === 'function') return window.marked(String(markdown || ''));
      return String(markdown || '')
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/^### (.*)$/gm, '<h3>$1</h3>').replace(/^## (.*)$/gm, '<h2>$1</h2>').replace(/^# (.*)$/gm, '<h1>$1</h1>')
        .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>').replace(/\*(.+?)\*/g, '<em>$1</em>')
        .replace(/\n\n+/g, '</p><p>').replace(/\n/g, '<br>')
        .replace(/^/, '<p>').replace(/$/, '</p>');
    };

    window.htmlToMarkdown = function (html) {
      if (window.TurndownService) {
        try {
          return new window.TurndownService().turndown(String(html || ''));
        } catch (error) {
          return String(html || '').replace(/<[^>]+>/g, '');
        }
      }
      return String(html || '').replace(/<[^>]+>/g, '').trim();
    };

    // 确认弹窗与原因填写弹窗共用的外壳：原生 <dialog> 自带焦圈、Esc 关闭和 backdrop。
    function openSiteModal(options) {
      // 同一时间只允许一个确认弹窗，避免连续点击叠出多层对话框。
      if (document.querySelector('[data-site-confirm-dialog]')) return null;
      options = options || {};
      const danger = options.danger === true;
      const previousFocus = document.activeElement;
      const dialog = document.createElement('dialog');
      dialog.className = 'site-confirm-overlay' + (danger ? ' site-confirm-overlay--danger' : '');
      dialog.setAttribute('data-site-confirm-dialog', '');
      dialog.setAttribute('aria-labelledby', 'site-confirm-title');
      dialog.setAttribute('aria-describedby', 'site-confirm-description');
      if (danger) dialog.setAttribute('role', 'alertdialog');
      dialog.innerHTML = '<div class="site-confirm-card"><span class="site-confirm-card__mark" aria-hidden="true"></span><span class="site-confirm-card__eyebrow"></span><h2 id="site-confirm-title"></h2><p id="site-confirm-description"></p><div class="site-confirm-verification" hidden><label for="site-confirm-input"></label><input id="site-confirm-input" class="form-control" type="text" autocomplete="off" spellcheck="false"><textarea id="site-confirm-textarea" class="form-control" rows="3" hidden></textarea></div><div class="site-confirm-actions"><button type="button" class="button button-secondary" data-confirm-cancel>取消</button><button type="button" class="button" data-confirm-ok></button></div></div>';
      dialog.querySelector('.site-confirm-card__mark').textContent = options.mark || (danger ? '!' : '?');
      dialog.querySelector('.site-confirm-card__eyebrow').textContent = options.eyebrow || (danger ? '高风险操作' : '操作确认');
      dialog.querySelector('h2').textContent = options.title || '请确认操作';
      dialog.querySelector('p').textContent = options.message || '确定继续吗？';
      const cancel = dialog.querySelector('[data-confirm-cancel]');
      const confirm = dialog.querySelector('[data-confirm-ok]');
      confirm.classList.add(danger ? 'button-danger' : 'button-primary');
      confirm.textContent = options.confirmLabel || (danger ? '确认执行' : '确认');
      let closed = false;
      const close = function () {
        if (closed) return;
        closed = true;
        window.WorkspaceMotion.close(dialog).then(function(){dialog.remove();});
        if (previousFocus && previousFocus.isConnected) previousFocus.focus({ preventScroll: true });
      };
      cancel.addEventListener('click', close);
      dialog.addEventListener('cancel', function (event) { event.preventDefault(); close(); });
      dialog.addEventListener('click', function (event) { if (event.target === dialog) close(); });
      dialog.addEventListener('close', close);
      document.body.appendChild(dialog);
      window.WorkspaceMotion.open(dialog);
      return { dialog: dialog, cancel: cancel, confirm: confirm, close: close, verification: dialog.querySelector('.site-confirm-verification') };
    }

    window.siteConfirm = function (message, onConfirm, options) {
      options = options || {};
      const modal = openSiteModal({
        message: message,
        title: options.title,
        confirmLabel: options.confirmLabel,
        danger: options.danger === true,
      });
      if (!modal) return;
      const input = modal.dialog.querySelector('#site-confirm-input');
      if (options.requireText) {
        modal.verification.hidden = false;
        modal.dialog.querySelector('label').textContent = '请输入“' + options.requireText + '”以确认';
        modal.confirm.disabled = true;
        input.addEventListener('input', function () { modal.confirm.disabled = input.value !== options.requireText; });
      }
      modal.confirm.addEventListener('click', function () {
        if (options.requireText && input.value !== options.requireText) return;
        modal.close();
        onConfirm();
      });
      if (options.requireText) input.focus();
      else modal.cancel.focus();
    };

    // 原因/备注类输入统一走站内弹窗，返回 Promise 式的回调结果，避免 window.prompt。
    window.sitePrompt = function (message, onSubmit, options) {
      options = options || {};
      const multiline = options.multiline !== false;
      const modal = openSiteModal({
        message: message,
        title: options.title || '请填写内容',
        confirmLabel: options.confirmLabel || '确认提交',
        danger: options.danger === true,
        eyebrow: options.danger === true ? '高风险操作' : '需要填写',
      });
      if (!modal) return;
      modal.verification.hidden = false;
      const label = modal.dialog.querySelector('label');
      label.textContent = options.label || '说明';
      const input = modal.dialog.querySelector(multiline ? '#site-confirm-textarea' : '#site-confirm-input');
      const unusedInput = modal.dialog.querySelector(multiline ? '#site-confirm-input' : '#site-confirm-textarea');
      label.setAttribute('for', input.id);
      unusedInput.remove();
      input.hidden = false;
      input.value = String(options.value || '');
      input.maxLength = Math.max(1, Number(options.maxLength) || 300);
      input.placeholder = options.placeholder || '';
      if (options.presets && options.presets.length) {
        const select = document.createElement('select');
        select.className = 'form-control reason-preset-select';
        select.setAttribute('data-reason-preset-for', input.id);
        select.setAttribute('aria-label', '选择常见原因');
        select.style.marginBottom = '8px';
        const placeholder = document.createElement('option');
        placeholder.value = '';
        placeholder.textContent = '选择常见原因…';
        select.appendChild(placeholder);
        options.presets.forEach(function (preset) {
          const opt = document.createElement('option');
          opt.value = preset;
          opt.textContent = preset;
          select.appendChild(opt);
        });
        input.insertAdjacentElement('beforebegin', select);
      }
      const required = options.required !== false;
      const sync = function () { modal.confirm.disabled = required && !input.value.trim(); };
      input.addEventListener('input', sync);
      sync();
      modal.confirm.addEventListener('click', function () {
        const value = input.value.trim();
        if (required && !value) return;
        modal.close();
        onSubmit(value);
      });
      input.focus();
    };

    // 授权只对确认后同步触发的这一次提交生效；取消、校验失败或重试都会重新确认。
    const confirmedForms = new WeakSet();
    // 「选择常见原因」下拉框：选中即把值填进目标字段，仍可继续手动编辑。
    document.addEventListener('change', function (event) {
      const select = event.target.closest && event.target.closest('[data-reason-preset-for]');
      if (!select) return;
      const target = document.getElementById(select.dataset.reasonPresetFor);
      if (target && select.value) {
        target.value = select.value;
        target.dispatchEvent(new Event('input', { bubbles: true }));
        target.focus();
      }
      select.value = '';
    });

    function replayAuthorizedSubmit(form, submitter) {
      if (!form.isConnected) return;
      confirmedForms.add(form);
      try { form.requestSubmit(submitter && submitter.form === form ? submitter : undefined); }
      finally { confirmedForms.delete(form); }
    }
    document.addEventListener('submit', function (event) {
      const form = event.target;
      if (!(form instanceof HTMLFormElement) || confirmedForms.has(form)) return;
      const submitter = event.submitter;
      // 需要填写原因的高危操作：先弹出输入框，确认后把值写入隐藏字段再提交。
      if (form.dataset.sitePrompt) {
        event.preventDefault();
        event.stopImmediatePropagation();
        window.sitePrompt(form.dataset.sitePrompt, function (value) {
          const fieldName = form.dataset.promptName || 'reason';
          let field = form.querySelector('input[type="hidden"][name="' + fieldName + '"]') || form.querySelector('input[type="hidden"][data-site-prompt-value]');
          if (!field) {
            field = document.createElement('input');
            field.type = 'hidden';
            field.setAttribute('data-site-prompt-value', '');
            form.appendChild(field);
          }
          field.name = fieldName;
          field.value = value;
          replayAuthorizedSubmit(form, submitter);
        }, {
          danger: form.dataset.promptVariant === 'danger',
          title: form.dataset.promptTitle,
          label: form.dataset.promptLabel,
          placeholder: form.dataset.promptPlaceholder,
          confirmLabel: form.dataset.promptConfirmLabel,
          multiline: form.dataset.promptMultiline !== 'false',
          required: form.dataset.promptRequired !== 'false',
          maxLength: form.dataset.promptMaxlength,
          presets: form.dataset.promptPresets ? JSON.parse(form.dataset.promptPresets) : undefined,
        });
        return;
      }
      if (!form.dataset.siteConfirm) return;
      event.preventDefault();
      event.stopImmediatePropagation();
      window.siteConfirm(form.dataset.siteConfirm, function () {
        replayAuthorizedSubmit(form, submitter);
      }, {
        danger: form.dataset.confirmVariant === 'danger',
        title: form.dataset.confirmTitle,
        confirmLabel: form.dataset.confirmLabel,
        requireText: form.dataset.confirmText,
      });
    }, true);

    document.addEventListener('DOMContentLoaded', function () {
      let pageTitle = document.title;
      const notifiedStorageKey = 'rent-notified-notification-ids';
      const updateBrowserTitle = function (count) {
        document.title = count > 0 ? pageTitle + ' (' + count + ')' : pageTitle;
      };
      const showNotification = function (notification, unreadCount) {
        const overlay = document.createElement('div');
        overlay.className = 'notification-toast notification-toast--stacked';
        overlay.innerHTML = '<article class="notification-toast__card" role="status"><div class="notification-toast__header"><span class="notification-toast__mark">N</span><strong></strong><button type="button" aria-label="关闭通知">×</button></div><p></p><a class="button button-sm button-secondary" href="/notifications">查看通知</a></article>';
        overlay.querySelector('strong').textContent = notification.title || '新通知';
        overlay.querySelector('p').textContent = '你收到了 ' + unreadCount + ' 条新消息';
        const dismiss = function () { fetch('/notifications/' + encodeURIComponent(notification.id) + '/dismiss', { method: 'POST' }).finally(function () { overlay.remove(); }); };
        overlay.querySelector('.notification-toast__header button').addEventListener('click', dismiss);
        getNoticeLayer().appendChild(overlay);
      };
      const pollNotifications = function () {
        fetch('/notifications/unread', { headers: { Accept: 'application/json' }, credentials: 'same-origin' }).then(function (response) { return response.ok ? response.json() : null; }).then(function (payload) {
          const notifications = payload && Array.isArray(payload.notifications) ? payload.notifications : [];
          updateBrowserTitle(notifications.length);
          if (!notifications.length) return;
          const currentIds = notifications.map(function (item) { return String(item.id); });
          let storedIds = [];
          try { storedIds = JSON.parse(localStorage.getItem(notifiedStorageKey) || '[]'); } catch (e) { storedIds = []; }
          if (!Array.isArray(storedIds)) storedIds = [];
          const newNotifications = notifications.filter(function (item) { return item.type !== 'announcement' && storedIds.indexOf(String(item.id)) === -1; });
          if (newNotifications.length) {
            const notification = newNotifications[0];
            if ('Notification' in window && Notification.permission === 'granted') {
              new Notification(notification.title || '新通知', { body: notification.message || '你收到了一条新通知' });
            }
            showNotification(notification, notifications.length);
          }
          const knownIds = storedIds.concat(currentIds).filter(function (id, index, ids) { return ids.indexOf(id) === index; });
          try { localStorage.setItem(notifiedStorageKey, JSON.stringify(knownIds.slice(-200))); } catch (e) {}
        }).catch(function () {});
      };
      // Notification permission is requested only after a user interaction.
      document.addEventListener('click', function requestNotificationPermission() {
        if ('Notification' in window && Notification.permission === 'default') Notification.requestPermission();
        document.removeEventListener('click', requestNotificationPermission);
      }, { once: true });
      window.setTimeout(function () { pollNotifications(); window.setInterval(pollNotifications, 60000); }, 2500);
    });

    document.addEventListener('DOMContentLoaded', function () {
      document.querySelectorAll('table:not(.official-document__table)').forEach(function (table) {
        const labels = Array.from(table.querySelectorAll('thead th')).map(function (header) { return header.textContent.trim(); });
        table.querySelectorAll('tbody tr').forEach(function (row) {
          Array.from(row.children).forEach(function (cell, index) {
            if (cell instanceof HTMLElement) cell.dataset.label = labels[index] || '';
          });
        });
      });
    });

    // 全局工作台搜索：功能入口来自当前角色可见的侧边栏，订单由权限受控的 API 返回。
    document.addEventListener('DOMContentLoaded', function () {
      const search = document.getElementById('global-search');
      const input = document.getElementById('global-search-input');
      const results = document.getElementById('global-search-results');
      const meta = document.getElementById('global-search-meta');
      const triggers = Array.from(document.querySelectorAll('[data-global-search-trigger]'));
      if (!search || !input || !results || !meta || !triggers.length) return;

      let activeIndex = -1;
      let debounceTimer = null;
      let requestToken = 0;
      let resultLinks = [];
      let previousFocus = null;

      const getPageEntries = function () {
        const seen = new Set();
        return Array.from(document.querySelectorAll('.sidebar-section a, #global-search-index a')).map(function (link) {
          const href = link.getAttribute('href') || '';
          const label = (link.textContent || '').replace(/\s+/g, ' ').trim();
          const group = link.dataset.searchGroup || link.closest('details')?.querySelector('summary')?.textContent.replace(/\s+/g, ' ').trim() || '导航';
          const keywords = link.dataset.searchKeywords || '';
          const key = href + '|' + label;
          if (!href || !label || seen.has(key)) return null;
          seen.add(key);
          return { href: href, label: label, group: group, keywords: keywords, icon: link.querySelector('.nav-icon')?.innerHTML || '' };
        }).filter(Boolean);
      };

      const matches = function (entry, query) {
        if (!query) return true;
        const haystack = (entry.label + ' ' + entry.group + ' ' + (entry.keywords || '') + ' ' + entry.href).toLowerCase();
        return haystack.indexOf(query.toLowerCase()) !== -1;
      };

      const appendHeading = function (label, count) {
        const heading = document.createElement('div');
        heading.className = 'global-search__section-title';
        heading.textContent = label + (count ? ' · ' + count : '');
        results.appendChild(heading);
      };

      const appendResult = function (item, type, index) {
        const link = document.createElement('a');
        link.className = 'global-search__result';
        link.href = item.href;
        link.id = 'global-search-option-' + index;
        link.setAttribute('role', 'option');
        link.setAttribute('aria-selected', 'false');
        const icon = document.createElement('span');
        icon.className = 'global-search__result-icon' + (type === 'order' ? ' is-order' : '');
        if (item.icon) icon.innerHTML = item.icon;
        else icon.textContent = type === 'order' ? 'OD' : '↗';
        const copy = document.createElement('span');
        copy.className = 'global-search__result-copy';
        const title = document.createElement('strong');
        title.textContent = item.title;
        const detail = document.createElement('small');
        detail.textContent = item.detail || '';
        copy.append(title, detail);
        const arrow = document.createElement('span');
        arrow.className = 'global-search__result-arrow';
        arrow.textContent = '↗';
        link.append(icon, copy, arrow);
        link.addEventListener('click', function () { close(); });
        results.appendChild(link);
        resultLinks.push(link);
      };

      const render = function (query, orders, loading, documents = []) {
        const normalizedQuery = query.trim();
        const pages = getPageEntries().filter(function (entry) { return matches(entry, normalizedQuery); });
        resultLinks = [];
        activeIndex = -1;
        results.replaceChildren();
        input.removeAttribute('aria-activedescendant');

        if (!normalizedQuery) {
          const common = pages.filter(function (entry) { return entry.href.indexOf('/dashboard') !== -1 || entry.href.indexOf('/orders') !== -1 || entry.href.indexOf('/devices') !== -1; }).slice(0, 8);
          appendHeading('常用功能', common.length);
          common.forEach(function (entry) { appendResult({ href: entry.href, title: entry.label, detail: entry.group, icon: entry.icon }, 'page', resultLinks.length); });
          meta.textContent = '输入名称、订单号、单据号或关键词开始搜索';
        } else {
          if (pages.length) {
            appendHeading('功能入口', pages.length);
            pages.forEach(function (entry) { appendResult({ href: entry.href, title: entry.label, detail: entry.group, icon: entry.icon }, 'page', resultLinks.length); });
          }
          if (loading) {
            appendHeading('订单', '…');
            const loadingRow = document.createElement('p');
            loadingRow.className = 'global-search__empty';
            loadingRow.textContent = '正在查找订单…';
            results.appendChild(loadingRow);
          } else if (orders.length) {
            appendHeading('订单', orders.length);
            orders.forEach(function (order) {
              appendResult({ href: order.target, title: order.orderNo, detail: [order.customerName, order.deviceName, order.statusLabel].filter(Boolean).join(' · ') }, 'order', resultLinks.length);
            });
          }
          if (documents.length) {
            appendHeading('单据', documents.length);
            documents.forEach(function (document) {
              appendResult({ href: document.target, title: document.number, detail: [document.typeLabel, document.orderNo, document.customerName, document.statusLabel].filter(Boolean).join(' · ') }, 'document', resultLinks.length);
            });
          }
          if (!pages.length && !orders.length && !documents.length && !loading) {
            const empty = document.createElement('p');
            empty.className = 'global-search__empty';
            empty.textContent = '没有找到匹配内容，试试订单号、单据号或客户名。';
            results.appendChild(empty);
          }
          meta.textContent = loading ? '正在搜索工作台…' : (pages.length + orders.length + documents.length ? '找到 ' + (pages.length + orders.length + documents.length) + ' 个结果' : '没有匹配结果');
        }
      };

      const setActive = function (nextIndex) {
        if (!resultLinks.length) return;
        activeIndex = (nextIndex + resultLinks.length) % resultLinks.length;
        resultLinks.forEach(function (link, index) {
          const selected = index === activeIndex;
          link.classList.toggle('is-selected', selected);
          link.setAttribute('aria-selected', String(selected));
        });
        input.setAttribute('aria-activedescendant', resultLinks[activeIndex].id);
      };

      const open = function (event) {
        if (event) event.preventDefault();
        previousFocus = document.activeElement;
        search.setAttribute('aria-hidden', 'false');
        document.body.classList.add('is-global-search-open');
        document.body.classList.remove('is-nav-open');
        document.querySelector('.mobile-nav-toggle')?.setAttribute('aria-expanded', 'false');
        document.querySelector('.mobile-nav-scrim')?.setAttribute('aria-hidden', 'true');
        render(input.value, [], false);
        window.requestAnimationFrame(function () { input.focus(); input.select(); });
      };

      function close(event) {
        if (event) event.preventDefault();
        search.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('is-global-search-open');
        if (previousFocus instanceof HTMLElement) previousFocus.focus();
      }

      triggers.forEach(function (trigger) { trigger.addEventListener('click', open); });
      search.querySelectorAll('[data-global-search-close]').forEach(function (element) { element.addEventListener('click', close); });
      document.addEventListener('click', function (event) {
        const target = event.target instanceof Element ? event.target : null;
        if (target?.closest('.global-search__result') && search.getAttribute('aria-hidden') !== 'true') close();
      }, true);
      input.addEventListener('input', function () {
        const query = input.value.trim();
        window.clearTimeout(debounceTimer);
        render(query, [], query.length >= 2);
        if (query.length < 2) return;
        const token = ++requestToken;
        debounceTimer = window.setTimeout(function () {
          fetch('/api/search?q=' + encodeURIComponent(query), { headers: { Accept: 'application/json' }, credentials: 'same-origin' })
            .then(function (response) { return response.ok ? response.json() : Promise.reject(new Error('search failed')); })
            .then(function (payload) {
              if (token !== requestToken || input.value.trim() !== query) return;
              render(query, Array.isArray(payload.orders) ? payload.orders : [], false, Array.isArray(payload.documents) ? payload.documents : []);
            })
            .catch(function () {
              if (token !== requestToken || input.value.trim() !== query) return;
              render(query, [], false);
              meta.textContent = '订单搜索暂时不可用，仍可使用功能入口';
            });
        }, 180);
      });
      input.addEventListener('keydown', function (event) {
        if (event.key === 'ArrowDown') { event.preventDefault(); setActive(activeIndex + 1); }
        else if (event.key === 'ArrowUp') { event.preventDefault(); setActive(activeIndex - 1); }
        else if (event.key === 'Enter' && activeIndex >= 0 && resultLinks[activeIndex]) { event.preventDefault(); resultLinks[activeIndex].click(); }
        else if (event.key === 'Escape') { close(event); }
      });
      document.addEventListener('keydown', function (event) {
        if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === 'k') { event.preventDefault(); if (search.getAttribute('aria-hidden') === 'true') open(event); else close(event); }
        else if (event.key === 'Escape' && search.getAttribute('aria-hidden') !== 'true') close(event);
      });
    });

    // 通告源码自动换行并随内容展开；折叠历史记录打开时重新测量。
    document.addEventListener('DOMContentLoaded', function () {
      const contentSizing = window.CSS && window.CSS.supports('field-sizing', 'content');
      const resize = function (area) { if (!area.getClientRects().length) return; area.style.height = 'auto'; if (!contentSizing) area.style.height = (area.scrollHeight + 2) + 'px'; };
      const resizeAll = function () { document.querySelectorAll('textarea.announcement-source').forEach(resize); };
      document.addEventListener('input', function (event) { if (event.target.matches('textarea.announcement-source')) resize(event.target); });
      document.addEventListener('toggle', resizeAll, true);
      document.addEventListener('site:content-updated', resizeAll);
      window.addEventListener('resize', resizeAll);
      resizeAll();
    });

    // 移动端导航抽屉：汉堡按钮滑出侧栏，点遮罩 / 链接 / Esc 关闭。
    document.addEventListener('DOMContentLoaded', function () {
      const body = document.body;
      const toggle = document.querySelector('.mobile-nav-toggle');
      const scrim = document.querySelector('.mobile-nav-scrim');
      const sidebar = document.getElementById('app-sidebar');

        body.classList.toggle('is-nav-open', open);
        toggle.setAttribute('aria-expanded', String(open));
        if (scrim) scrim.setAttribute('aria-hidden', String(!open));
      };
      toggle.addEventListener('click', function () { setOpen(!body.classList.contains('is-nav-open')); });
      if (scrim) scrim.addEventListener('click', function () { setOpen(false); });
      sidebar.addEventListener('click', function (event) { if (event.target.closest('a')) setOpen(false); });
      document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape' && body.classList.contains('is-nav-open')) setOpen(false);
      });
    });

    // 账户与页脚折叠菜单：点击菜单外或按 Esc 时收起。
    // 用 pointerdown + 捕获阶段监听，确保在 SPA 局部导航处理器

        document.addEventListener('keydown', function (event) {
          if (event.key === 'Escape' && more.open) {
            more.open = false;
            const summary = more.querySelector('summary');
            if (summary) summary.focus();
          }
        });
      });
    });

        editor.style.pointerEvents = 'auto';

        editor.style.visibility = 'visible';
        editor.style.opacity = '1';
        editor.setAttribute('spellcheck', 'false');
        const toolbar = document.createElement('div');
        toolbar.className = 'markdown-editor-toolbar';
        [['标题', '# '], ['粗体', '**文字**'], ['斜体', '*文字*'], ['列表', '- 项目'], ['分隔线', '\n---\n']].forEach(function (item) {
          const button = document.createElement('button');
          button.type = 'button';
          button.textContent = item[0];
          button.addEventListener('click', function () {
            const start = editor.selectionStart || 0;
            const end = editor.selectionEnd || start;
            const selected = editor.value.slice(start, end) || item[1];
            editor.setRangeText(item[0] === '标题' ? '# ' + selected : item[1].includes('文字') ? item[1].replace('文字', selected) : item[1], start, end, 'end');
            editor.dispatchEvent(new Event('input', { bubbles: true }));
            editor.focus();
          });
          toolbar.appendChild(button);
        });
        editor.parentNode.insertBefore(toolbar, editor);
      });
    });

    (function () {
      const current = new URL(window.location.href);
      document.querySelectorAll('.sidebar-section, .mobile-bottom-nav').forEach(function (nav) {
        const links = Array.from(nav.querySelectorAll('a[href]'));
        links.forEach(function (link) { link.classList.remove('active'); link.removeAttribute('aria-current'); });
        const active = links.filter(function (link) {
          const url = new URL(link.href, current);
          const path = url.pathname.replace(/\/$/, '') || '/';
          if (url.origin !== current.origin || !(current.pathname === path || (path !== '/' && current.pathname.startsWith(path + '/')))) return false;
          return Array.from(url.searchParams).every(function (entry) { return current.searchParams.get(entry[0]) === entry[1]; });
        }).sort(function (a, b) { return b.getAttribute('href').length - a.getAttribute('href').length; })[0];
        if (active) {
          active.classList.add('active');
          active.setAttribute('aria-current', 'page');
          const group = active.closest('details.sidebar-nav-group');
          if (group) group.open = true;
        }
      });
    })();

        message.dataset.previewReady = '1';
        const full = message.innerHTML;
        const text = (message.textContent || '').replace(/\s+/g, ' ').trim();
        if (text.length <= 120) return;
        message.textContent = text.slice(0, 120) + '…';
        const details = document.createElement('details');
        details.className = 'notification-detail';
        const summary = document.createElement('summary');
        summary.textContent = '查看详情';
        const content = document.createElement('div');
        content.className = 'notification-message';
        content.innerHTML = full;
        details.append(summary, content);
        message.after(details);
      });
    });
